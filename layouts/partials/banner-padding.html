{{- /* Returns the element's class attributes with the top padding that clears the fixed nav and the active banners. The webinar banner hides itself client-side. When it is the only strip, the padding assumes the strip, which is the state before Alpine runs, and an Alpine :class on the body's webinar state swaps in the no-strip padding once the banner reports none. Context: "context" (the page), "classes" (the element's other classes), "base", and optionally "thinStrip", "anniversary", and "both", which default to a flush hero's clearance. */ -}} {{- $showWebinarBanner := .context.Scratch.Get "webinarBannerActive" -}} {{- $showSaleBanner := .context.Scratch.Get "saleBannerActive" -}} {{- $showAnniversaryBanner := .context.Scratch.Get "anniversaryBannerActive" -}} {{- $showThinStrip := or $showWebinarBanner $showSaleBanner -}} {{- $thinStrip := .thinStrip | default "pt-28 md:pt-28" -}} {{- $anniversary := .anniversary | default "pt-40 md:pt-32" -}} {{- $both := .both | default "pt-40 md:pt-40" -}} {{- $padding := .base -}} {{- if and $showThinStrip $showAnniversaryBanner -}} {{- $padding = $both -}} {{- else if $showThinStrip -}} {{- $padding = $thinStrip -}} {{- else if $showAnniversaryBanner -}} {{- $padding = $anniversary -}} {{- end -}} {{- $attrs := printf "class=\"%s %s\"" .classes $padding -}} {{- if and $showWebinarBanner (not $showSaleBanner) -}} {{- $paddingNoStrip := cond $showAnniversaryBanner $anniversary .base -}} {{- /* Only the tokens that differ enter the Alpine object, so a token both paddings share stays in the static class and is never removed. */ -}} {{- $onlyNoStrip := delimit (complement (split $padding " ") (split $paddingNoStrip " ")) " " -}} {{- $onlyStrip := delimit (complement (split $paddingNoStrip " ") (split $padding " ")) " " -}} {{- $attrs = printf `%s :class="{ '%s': webinar.state === 'none', '%s': webinar.state !== 'none' }"` $attrs $onlyNoStrip $onlyStrip -}} {{- end -}} {{- return ($attrs | safeHTMLAttr) -}}
